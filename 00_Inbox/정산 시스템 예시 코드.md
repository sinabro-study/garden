## 2. 정산 원천 — 불변 엔티티
```kotlin
@Entity
@Table(
    name = "settlement_source",
    uniqueConstraints = [UniqueConstraint(name = "uk_source_event_id", columnNames = ["event_id"])],
)
class SettlementSource(
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    val id: Long = 0L,

    // Kafka 이벤트 고유 ID — 유니크 제약이 중복 소비의 최종 방어선
    @Column(name = "event_id", nullable = false, updatable = false)
    val eventId: String,

    @Column(nullable = false, updatable = false)
    val orderId: String,

    @Enumerated(EnumType.STRING)
    @Column(nullable = false, updatable = false)
    val txType: TxType,

    // 취소/환불은 음수 금액의 별도 레코드로 적재 (원 레코드 수정 금지)
    @Column(nullable = false, updatable = false, precision = 15, scale = 0)
    val amount: BigDecimal,

    @Column(nullable = false, updatable = false)
    val settlementBaseDate: LocalDate,

    @Column(nullable = false, updatable = false)
    val occurredAt: LocalDateTime,
)

enum class TxType { PAYMENT, CANCEL, REFUND }
```
- 모든 컬럼 `updatable = false` — JPA 레벨에서도 불변 강제
- 금액은 `BigDecimal`(원 단위, scale 0) — Double/Float 금지 원칙의 코드 표현

## 3. 원천 수집기 — Kafka 소비 + 멱등 처리
```kotlin
@Component
class SettlementSourceCollector(
    private val sourceRepository: SettlementSourceRepository,
) {
    @KafkaListener(topics = ["commerce.payment-event"], groupId = "settlement-collector")
    @Transactional
    fun consume(event: PaymentEvent) {
        // 1차: 조회로 중복 차단 (릴레이는 at-least-once — 중복 발행 전제)
        if (sourceRepository.existsByEventId(event.eventId)) {
            return
        }
        try {
            sourceRepository.save(event.toSettlementSource())
        } catch (e: DataIntegrityViolationException) {
            // 2차: 동시 소비 레이스는 유니크 제약이 차단 — 중복이므로 정상 종료
            log.info { "duplicate event skipped: ${event.eventId}" }
        }
    }
}
```
- 멱등 2계층: 조회(빠른 경로) + 유니크 제약(레이스 방어) — 어느 쪽이 뚫려도 중복 적재 불가

## 4. 회차 상태 CAS — 이중 실행 방어의 핵심
```kotlin
interface SettlementRepository : JpaRepository<Settlement, Long> {
    @Modifying
    @Query(
        """
        update Settlement s
           set s.status = :next
         where s.id = :id
           and s.status = :expected
        """
    )
    fun compareAndSetStatus(id: Long, expected: SettlementStatus, next: SettlementStatus): Int
}
```
```kotlin
@Transactional
fun startAggregation(settlementId: Long) {
    val updated = settlementRepository.compareAndSetStatus(
        id = settlementId,
        expected = SettlementStatus.CREATED,
        next = SettlementStatus.AGGREGATING,
    )
    // 갱신 0건 = 다른 인스턴스가 이미 선점 — 락이 뚫려도 여기서 중단됨
    check(updated == 1) { "settlement $settlementId is already in progress" }
}
```
- ShedLock이 만료로 뚫려도 이 조건부 UPDATE는 DB 원자성으로 한쪽만 통과 — 정확성의 최종 보루

## 5. 정산 배치 — ShedLock 스케줄러와 계산 Step
```kotlin
@Component
class SettlementScheduler(
    private val jobLauncher: JobLauncher,
    private val dailySettlementJob: Job,
) {
    @Scheduled(cron = "0 0 4 * * *")
    @SchedulerLock(name = "daily-settlement", lockAtMostFor = "PT4H", lockAtLeastFor = "PT10M")
    fun run() {
        val params = JobParametersBuilder()
            .addLocalDate("baseDate", LocalDate.now().minusDays(1))
            .toJobParameters()
        // 동일 파라미터 재실행은 JobRepository가 차단 — 부가 방어 계층
        jobLauncher.run(dailySettlementJob, params)
    }
}
```
```kotlin
@Configuration
class CalculateStepConfig(
    private val feeCalculator: FeeCalculator,
) {
    @Bean
    fun calculateStep(
        jobRepository: JobRepository,
        txManager: PlatformTransactionManager,
        sourceReader: RepositoryItemReader<SettlementSource>,
        detailWriter: RepositoryItemWriter<SettlementDetail>,
    ): Step =
        StepBuilder("calculateStep", jobRepository)
            .chunk<SettlementSource, SettlementDetail>(1_000, txManager)
            .reader(sourceReader) // 페이징 조회 — 전건 메모리 적재 금지
            .processor { source -> feeCalculator.calculate(source) }
            .writer(detailWriter)
            .build()
}
```
```kotlin
// settlement-domain — 외부 의존성 없는 순수 계산 로직 (단위 테스트 집중 영역)
class FeeCalculator(
    private val feePolicyProvider: FeePolicyProvider,
) {
    fun calculate(source: SettlementSource): SettlementDetail {
        // 요율은 거래 발생 시점 스냅샷 사용 — 정산 시점 재조회 금지
        val policy = feePolicyProvider.snapshotAt(source.orderId, source.occurredAt)

        val platformFee = source.amount
            .multiply(policy.feeRate)
            .setScale(0, RoundingMode.DOWN) // 반올림 정책: 원 단위 절사 — 전 시스템 단일 규칙

        return SettlementDetail(
            sourceId = source.id,
            amount = source.amount,
            platformFee = platformFee,
            appliedFeeRate = policy.feeRate, // 적용 요율 스냅샷 저장 (역추적 근거)
            payoutAmount = source.amount.subtract(platformFee),
        )
    }
}
```

## 6. 합계 검증 Step — 확정 차단 게이트
```kotlin
@Component
class VerifyAggregationTasklet(
    private val detailRepository: SettlementDetailRepository,
    private val settlementRepository: SettlementRepository,
) : Tasklet {
    override fun execute(contribution: StepContribution, chunkContext: ChunkContext): RepeatStatus {
        val settlementId = chunkContext.jobParameter<Long>("settlementId")
        val settlement = settlementRepository.findByIdOrNull(settlementId)
            ?: error("settlement not found: $settlementId")

        val detailSum = detailRepository.sumPayoutAmountBySettlementId(settlementId)
        if (detailSum != settlement.aggregatedAmount) {
            // 건별 합 ≠ 집계값 — 확정 진입 차단
            settlement.hold(reason = "AGGREGATION_MISMATCH detail=$detailSum aggregated=${settlement.aggregatedAmount}")
            return RepeatStatus.FINISHED
        }
        settlement.verify()
        return RepeatStatus.FINISHED
    }
}
```

## 7. 지급 모듈 — 멱등키와 타임아웃 처리
```kotlin
@Service
class PayoutService(
    private val payoutRepository: PayoutRepository,
    private val bankingClient: BankingClient,
    private val ledgerService: LedgerService,
) {
    @Transactional
    fun execute(settlementId: Long) {
        // idempotency_key 유니크 제약 — 동일 회차 중복 지급 원천 차단 (방어 최종 계층)
        val payout = payoutRepository.findBySettlementId(settlementId)
            ?: payoutRepository.save(Payout.create(settlementId))

        if (payout.isCompleted()) return

        try {
            val result = bankingClient.transfer(
                TransferRequest(
                    idempotencyKey = payout.idempotencyKey,
                    account = payout.account,
                    amount = payout.amount,
                )
            )
            complete(payout, result)
        } catch (e: BankingTimeoutException) {
            // 임의 재시도 금지 — 이체가 됐는지 모르는 상태에서 재요청하면 중복 이체
            val status = bankingClient.getTransferResult(payout.idempotencyKey)
            when (status) {
                TransferStatus.SUCCESS -> complete(payout, status.toResult())
                TransferStatus.FAILED -> payout.fail(status.resultCode)
                TransferStatus.UNKNOWN -> payout.markPendingCheck() // 별도 배치로 재조회
            }
        }
    }

    private fun complete(payout: Payout, result: TransferResult) {
        payout.complete(result)
        ledgerService.recordPayout(payout) // 지급 완료 분개
    }
}
```

## 8. 원장 분개 — 복식부기
```kotlin
@Service
class LedgerService(
    private val ledgerRepository: LedgerEntryRepository,
) {
    fun recordPayout(payout: Payout) {
        // 차변·대변 쌍 기록 — Append-only, UPDATE/DELETE 금지
        ledgerRepository.saveAll(
            listOf(
                LedgerEntry.debit(Account.UNSETTLED_PAYABLE, payout.amount, payout.id),  // 미정산금(부채) 감소
                LedgerEntry.credit(Account.CASH, payout.amount, payout.id),              // 현금(자산) 감소
            )
        )
    }
}
```
- 모든 분개는 차변 합 = 대변 합 — 잔액 대사 시 이 불변식으로 전체 자금 흐름 검증
